cmake_minimum_required(VERSION 3.16)
project(TracePipelineSimulator VERSION 1.0.0 LANGUAGES CXX)
add_library(pipeline src/simulator.cpp)
target_include_directories(pipeline PUBLIC include)
target_compile_features(pipeline PUBLIC cxx_std_17)
set_target_properties(pipeline PROPERTIES CXX_EXTENSIONS OFF)
add_executable(proj2 src/main.cpp)
target_link_libraries(proj2 PRIVATE pipeline)
include(CTest)
if(BUILD_TESTING)
  add_executable(pipeline_tests tests/simulator_tests.cpp)
  target_link_libraries(pipeline_tests PRIVATE pipeline)
  add_test(NAME pipeline_unit_tests COMMAND pipeline_tests)
endif()
if(BUILD_TESTING)
  add_test(NAME pipeline_cli_sample COMMAND proj2 "${CMAKE_CURRENT_SOURCE_DIR}/tests/sample.trace" 1 5 2)
  set_tests_properties(pipeline_cli_sample PROPERTIES PASS_REGULAR_EXPRESSION "Retired: 5")
  add_test(NAME pipeline_cli_invalid COMMAND proj2 "${CMAKE_CURRENT_SOURCE_DIR}/tests/sample.trace" 1 5 5)
  set_tests_properties(pipeline_cli_invalid PROPERTIES WILL_FAIL TRUE)
endif()
